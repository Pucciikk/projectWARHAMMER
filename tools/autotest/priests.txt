# Техножрецы крупным планом: спереди, в три четверти, сзади и на ходу
cmd time set noon
cmd weather clear
cmd gamemode creative
cmd difficulty normal
cmd execute in minecraft:overworld run tp @s 0 150 -6 0 0
wait 40
cmd fill -16 149 -10 16 149 12 minecraft:smooth_stone
cmd fill -16 150 -10 16 160 12 minecraft:air
cmd kill @e[type=!minecraft:player]
cmd summon warpbound:tech_priest -2 150 0 {NoAI:1b,Rotation:[180f,0f]}
cmd summon warpbound:dark_tech_priest 2.5 150 0 {NoAI:1b,Rotation:[180f,0f]}
wait 30
hud off
cmd tp @s 0.25 150.8 -5.5 0 -5
wait 15
shot 01_front
wait 3
cmd tp @s -2 151.3 -2.2 0 10
wait 15
shot 02_face
wait 3
cmd tp @s 2.5 152.2 -2.8 0 12
wait 15
shot 02b_dark_face
wait 3
cmd tp @s -5.5 151 -3.5 -50 5
wait 15
shot 03_three_quarter
wait 3
cmd tp @s 0.25 151.5 5.5 180 8
wait 15
shot 04_back
wait 3
# на ходу: без NoAI жрецы бродят
cmd kill @e[type=!minecraft:player]
cmd summon warpbound:tech_priest -2 150 2
cmd summon warpbound:dark_tech_priest 3 150 2
wait 80
cmd tp @s 0.5 152 -6 0 8
wait 20
shot 05_walking
wait 3
hud on
